% Pedagogical listing roles / 教学型代码清单角色。
% Exact syntax is shown only when it carries the explanation; commands are
% visually quieter than programs. / 仅在语法本身参与论证时突出代码，命令则弱化呈现。
\lstdefinestyle{explanatory}{
  basicstyle=\ttfamily\footnotesize,
  columns=fullflexible,
  keepspaces=true,
  showstringspaces=false,
  breaklines=true,
  breakatwhitespace=false,
  frame=none,
  numbers=none,
  xleftmargin=0.6em,
  aboveskip=0.7\baselineskip,
  belowskip=0.5\baselineskip,
  keywordstyle=\color{blue!60!black},
  commentstyle=\color{green!35!black},
  stringstyle=\color{orange!50!black}
}
\lstdefinestyle{normative}{
  style=explanatory,
  basicstyle=\ttfamily\small,
  frame=single,
  framerule=0.35pt,
  rulecolor=\color{black!35},
  numbers=left,
  numberstyle=\footnotesize\color{black!55},
  xleftmargin=2.1em,
  framexleftmargin=1.6em
}
\lstdefinestyle{command}{
  style=explanatory,
  backgroundcolor=\color{black!4},
  rulecolor=\color{black!20},
  frame=single,
  framerule=0.25pt,
  keywordstyle=\color{black},

  stringstyle=\color{black}
}

% LLVM IR and RISC-V are intentionally lightweight lexical definitions.
% LLVM IR 与 RISC-V 仅定义论文所需的轻量词法高亮。
\lstdefinelanguage{LLVMIR}{
  morekeywords={define,declare,target,private,constant,global,align,ptr,readonly,
    entry,ret,br,label,phi,icmp,select,sext,getelementptr,inbounds,load,store,
    call,alloca,add,mul,void,i1,i8,i32,i64},
  sensitive=true,
  morecomment=[l]{;},
  morestring=[b]"
}
\lstdefinelanguage{RISCV}{
  morekeywords={add,addi,addw,addiw,sub,mul,mulw,slli,lw,sw,ld,sd,li,mv,
    blt,bge,ble,bgt,beq,bne,j,call,ret,globl,type,size,text,section,attribute},
  sensitive=true,
  morecomment=[l]{\#}
}
